function histSave(list) { localStorage.setItem(HIST_KEY, JSON.stringify(list)); }

function fmtDT(ts) {
  var d = new Date(ts);

function fmtRem(ms) {
  if (ms <= 0) return '已过期';
  if (ms < 3600000) return '不足 1 小时';
  var days = Math.floor(ms / 86400000), hours = Math.floor((ms % 86400000) / 3600000);
